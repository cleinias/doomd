\begin{tabularx}{\textwidth}{@{}ll>{\raggedright\arraybackslash}X@{}}
\toprule
\textbf{Panel} & \textbf{Key} & \textbf{What it does} \\
\midrule
Stack       & \key{RET}         & Select a frame: jump to its code and show its variables \\
Breakpoints & \key{RET}         & Go to the breakpoint; on the exception rows (raised / uncaught)
                                turn breaking on exceptions on or off \\
            & \key{D} / \key{d}   & Disable a breakpoint / delete it \\
            & \key{e}           & Edit a log breakpoint's message \\
Scope       & \key{e}           & Expand or collapse a dict, list or object \\
            & \key{W}           & Add the variable to Watch \\
            & \key{=}           & Change the variable's value while paused \\
Watch       & \key{W}           & Remove the expression from Watch \\
            & \key{C-x C-q}     & Edit the list of watch expressions as text \\
\bottomrule
\end{tabularx}
